```python
WORKFLOW_PY_PIN_RE = re.compile(r"(\bpython-version:\s*['\"])(\d+\.\d+\.\d+)(['\"])")
WORKFLOW_USES_PIN_RE = re.compile(
    r"^(\s*(?:-\s*)?uses:\s*)([A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+(?:/[A-Za-z0-9_.-]+)?)@"
    r"([0-9a-f]{40})(\s+#\s*v)(\d+(?:\.\d+){0,3})([^\r\n]*)$"
)
SMOKE_PY_PIN_RE = re.compile(r"(CPython |python-version: ')(\d+\.\d+\.\d+)")


def _parse_version_tuple(raw: str) -> tuple[int, ...]:
    return tuple(int(part) for part in raw.split("."))


def merge_workflow_pins_bytes(src_bytes: bytes, dst_bytes: bytes) -> bytes:
    """Keep monotonically newer `python-version` / SHA-pinned `uses:` lines in `dst_bytes`."""
    try:
        src_text = src_bytes.decode("utf-8")
        dst_text = dst_bytes.decode("utf-8")
    except UnicodeDecodeError:
        return src_bytes

    dst_py_versions = [m.group(2) for m in WORKFLOW_PY_PIN_RE.finditer(dst_text)]
    max_dst_py: tuple[int, ...] | None = None
    max_dst_py_str = ""
    for ver in dst_py_versions:
        parsed = _parse_version_tuple(ver)
        if max_dst_py is None or parsed > max_dst_py:
            max_dst_py = parsed
            max_dst_py_str = ver

    dst_action_pins: dict[str, tuple[tuple[int, ...], str, str]] = {}
    for raw_line in dst_text.splitlines():
        m = WORKFLOW_USES_PIN_RE.match(raw_line)
        if not m:
            continue
        action_slug, sha, ver_str = m.group(2), m.group(3), m.group(5)
        ver_tuple = _parse_version_tuple(ver_str)
        prev = dst_action_pins.get(action_slug)
        if prev is None or ver_tuple > prev[0]:
            dst_action_pins[action_slug] = (ver_tuple, sha, ver_str)

    if max_dst_py is None and not dst_action_pins:
        return src_bytes

    def _replace_py(match: re.Match[str]) -> str:
        src_ver = _parse_version_tuple(match.group(2))
        if max_dst_py is not None and max_dst_py > src_ver:
            return f"{match.group(1)}{max_dst_py_str}{match.group(3)}"
        return match.group(0)

    merged_lines: list[str] = []
    for line in src_text.splitlines(keepends=True):
        ending = ""
        body = line
        if line.endswith("\r\n"):
            body, ending = line[:-2], "\r\n"
        elif line.endswith("\n"):
            body, ending = line[:-1], "\n"
        m = WORKFLOW_USES_PIN_RE.match(body)
        if m:
            slug = m.group(2)
            src_ver = _parse_version_tuple(m.group(5))
            dst_pin = dst_action_pins.get(slug)
            if dst_pin is not None and dst_pin[0] > src_ver:
                body = f"{m.group(1)}{slug}@{dst_pin[1]}{m.group(4)}{dst_pin[2]}{m.group(6)}"
        body = WORKFLOW_PY_PIN_RE.sub(_replace_py, body)
        merged_lines.append(body + ending)

    return "".join(merged_lines).encode("utf-8")


def merge_smoke_python_pin_bytes(src_smoke_bytes: bytes, dst_ci_bytes: bytes) -> bytes:
    """Keep `tests/test_smoke.py`'s r39 CPython pin aligned with destination `ci.yml` when prod is newer."""
    try:
        smoke_text = src_smoke_bytes.decode("utf-8")
        ci_text = dst_ci_bytes.decode("utf-8")
    except UnicodeDecodeError:
        return src_smoke_bytes
    ci_match = WORKFLOW_PY_PIN_RE.search(ci_text)
    if not ci_match:
        return src_smoke_bytes
    dst_ver_str = ci_match.group(2)
    dst_ver = _parse_version_tuple(dst_ver_str)

    def _replace_smoke_py(match: re.Match[str]) -> str:
        if dst_ver > _parse_version_tuple(match.group(2)):
            return f"{match.group(1)}{dst_ver_str}"
        return match.group(0)

    return SMOKE_PY_PIN_RE.sub(_replace_smoke_py, smoke_text).encode("utf-8")
```
